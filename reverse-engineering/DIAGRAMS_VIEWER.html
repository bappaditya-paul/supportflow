flowchart TD
    subgraph Flow1 ["Customer Turn & AI Response"]
        V1["Customer submits message in Widget"] --> W1["POST /api/v1/conversations/messages"]
        W1 --> DB1["Insert conversationTimelineItem (public)"]
        W1 --> REDIS1["Publish timelineItemCreated to Redis Pub/Sub"]
        REDIS1 --> OP1["Render in Operator Dashboard"]
        W1 --> Q1["Enqueue Job in BullMQ 'ai-agent'"]
        
        Q1 --> AI1["Primary AI Worker picks up turn"]
        AI1 --> TYPING1["Emit conversationTyping: true over WS"]
        AI1 --> RAG1["Cosine Similarity Query against pgvector"]
        
        RAG1 --> CHECK1{"Similarity Score >= 0.75?"}
        CHECK1 -- Yes --> GEN1["LLM generates policy-grounded reply"]
        GEN1 --> DB2["Insert AI Reply Timeline Item"]
        DB2 --> STREAM1["Stream tokens to Widget over WebSocket"]
        GEN1 --> DEBOUNCE1["Schedule Background Job (30s delay)"]
        
        CHECK1 -- No --> CLARIFY1["Invoke tool: requestKnowledgeClarification"]
    end

    subgraph Flow2 ["Self-Learning Knowledge Loop"]
        CLARIFY1 --> DB3["Insert into knowledgeClarificationRequest"]
        DB3 --> NOTIF1["Alert Operator: 'Missing Knowledge Detail'"]
        CLARIFY1 --> FALLBACK1["AI responds: 'Connecting you with human specialist...'"]
        
        NOTIF1 --> OP_ANS["Operator enters verified answer in Dashboard"]
        OP_ANS --> TRPC_ANS["trpc.knowledgeClarification.answer"]
        TRPC_ANS --> CHUNK1["Split answer into 512-token text chunks"]
        CHUNK1 --> EMBED1["Compute embeddings via text-embedding-3-small"]
        EMBED1 --> PGVEC1["Store vector in chunk table (HNSW Index)"]
        PGVEC1 --> RESOLVE_REQ["Mark Clarification as Resolved"]
        RESOLVE_REQ --> FUTURE_QUERY["Future customer turns answered autonomously!"]
    end
